# ------------------------------------------------------------------------------
# 1. 全局物理参数配置
# ------------------------------------------------------------------------------
N_TOTAL = 11               # 11维图拉普拉斯矩阵
N_SPACE = 3                # 涌现的三维空间扇区
R_ROOT = 1.0               # 几何特征尺度
DELTA_LOOP_THEORY = 11 / 544.0  # 理论回路修正因子
EPSILON_LOOP = 0.007441    # 亚临界运行点

# ------------------------------------------------------------------------------
# 2. 核心数值函数库

def build_M(epsilon):
    """构造 11x11 非厄米图拉普拉斯算子 M"""
    M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    np.random.seed(42)  # 固定随机种子，确保实验 100% 可重复
    V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
    
    for i in range(N_TOTAL):
        for j in range(N_TOTAL):
            if i != j:
                M[i, j] = epsilon * abs(V[i, j])
                
    kappa = np.zeros(N_TOTAL, dtype=complex)
    kappa[:N_SPACE] = 0.0
    kappa[N_SPACE:] = 1.0 / (R_ROOT**2)
    
    for i in range(N_TOTAL):
        M[i, i] = -np.sum(M[:, i]) + kappa[i]
        
    return M

def compute_nonorthogonality_metric(epsilon):
    """计算双正交基底和非正交重叠度规 S"""
    M = build_M(epsilon)
    evals, eigvecs_right = linalg.eig(M)
    
    # 归一化右本征态
    for k in range(N_TOTAL):
        norm = linalg.norm(eigvecs_right[:, k])
        if norm > 0:
            eigvecs_right[:, k] /= norm
            
    # 按照空间扇区权重排序
    spatial_weights = np.sum(np.abs(eigvecs_right[:N_SPACE, :])**2, axis=0)
    sort_idx = np.argsort(spatial_weights)[::-1]
    eigvecs_right = eigvecs_right[:, sort_idx]
    
    # 构建左本征态 (双正交基底 L^\dagger R = I)
    eigvecs_left = linalg.inv(eigvecs_right).T.conj()
    S = eigvecs_right.T.conj() @ eigvecs_right
    
    return S, eigvecs_right, eigvecs_left, sort_idx

def get_collapse_operator(epsilon, gamma=1.0):
    """构造有效塌缩算子 M_collapse"""
    M = build_M(epsilon)
    _, eigvecs_right, eigvecs_left, _ = compute_nonorthogonality_metric(epsilon)
    
    P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for k in range(N_SPACE):
        P_meas += np.outer(eigvecs_right[:, k], eigvecs_left[:, k].conj())
        
    I_mat = np.eye(N_TOTAL, dtype=complex)
    Q_unobs = I_mat - P_meas
    gamma_eff = gamma * DELTA_LOOP_THEORY
    M_collapse = M - 1j * gamma_eff * Q_unobs
    
    return M_collapse, Q_unobs

def extract_kraus_operators(epsilon, t, gamma=1.0):
    """SVD 稳定化与 Kraus 算子提取（量子 Dilation 闭合验证）"""
    M_collapse, _ = get_collapse_operator(epsilon, gamma)
    U_raw = linalg.expm(-1j * M_collapse * t)
    
    P, S_vals, Qh = linalg.svd(U_raw)
    S_stabilized = np.minimum(S_vals, 1.0)
    U_stab = P @ np.diag(S_stabilized) @ Qh
    
    I_mat = np.eye(N_TOTAL, dtype=complex)
    D = I_mat - U_stab.conj().T @ U_stab
    
    eigvals, eigvecs = linalg.eigh(D)
    eigvals = np.maximum(eigvals, 0.0)
    
    threshold = 1e-14
    valid_idx = np.where(eigvals > threshold)[0]
    kraus_set = [U_stab]
    
    for rank_idx, idx in enumerate(valid_idx):
        val = eigvals[idx]
        vec = eigvecs[:, idx]
        e_k = np.zeros(N_TOTAL, dtype=complex)
        if rank_idx < N_TOTAL:
            e_k[rank_idx] = 1.0
        A_k = np.sqrt(val) * np.outer(e_k, vec.conj())
        kraus_set.append(A_k)
        
    return kraus_set

def verify_trace_preservation(kraus_set):
    """验证迹守恒条件 sum A_k^dagger A_k = I"""
    completeness_sum = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for A in kraus_set:
        completeness_sum += A.conj().T @ A
    I_mat = np.eye(N_TOTAL, dtype=complex)
    return linalg.norm(completeness_sum - I_mat)

def extract_physical_parameters_direct(epsilon, gamma=1.0):
    """直接提取时空扇区的频率 Omega 与衰减率 Gamma"""
    M = build_M(epsilon)
    _, eigvecs_right, eigvecs_left, _ = compute_nonorthogonality_metric(epsilon)
    
    P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for k in range(N_SPACE):
        P_meas += np.outer(eigvecs_right[:, k], eigvecs_left[:, k].conj())
        
    I_mat = np.eye(N_TOTAL, dtype=complex)
    Q_unobs = I_mat - P_meas
    gamma_eff = gamma * DELTA_LOOP_THEORY
    M_collapse = M - 1j * gamma_eff * Q_unobs
    
    evals_coll, evecs_coll = linalg.eig(M_collapse)
    space_overlap = np.zeros(N_TOTAL)
    
    for i in range(N_TOTAL):
        v = evecs_coll[:, i]
        proj = P_meas @ v
        space_overlap[i] = np.abs(np.vdot(v, proj))
        
    space_order = np.argsort(space_overlap)[::-1]
    space_evals = evals_coll[space_order[:N_SPACE]]
    
    Gamma_space = np.abs(np.imag(space_evals))
    Omega_space = np.abs(np.real(space_evals))
    
    return np.mean(Omega_space), np.mean(Gamma_space)

def estimate_laplacian_error(N_range):
    """计算离散-连续误差 E(N) = C / N^(2/3) (已修复零宽空格语法Bug)"""
    C = 1.25
    errors = [C / (N ** (2.0 / 3.0)) for N in N_range]
    return np.array(errors)
