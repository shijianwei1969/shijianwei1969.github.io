# ==================== 体系全局常量（P5/P6锁定的拓扑内生参数） ====================
N_TOTAL = 11          # 全域席位数
N_SPACE = 3           # 时空扇区维度
N_VACUUM = 8          # 真空背景扇区维度
BETA_K = 3/8 * np.exp(-1/187)   # 拓扑压缩比 ≈ 0.3729
EPSILON_C = 0.377     # EP3临界微扰参数
R_ROOT = 1.0          # 宇宙学红底零点尺度
DELTA_LOOP = 11 / 544 # 拓扑边界纠缠比 ≈ 0.02022

# 席位索引分配
idx_space = list(range(N_SPACE))
idx_vacuum = list(range(N_SPACE, N_TOTAL))

# ==================== 单元1：构建非厄米图拉普拉斯算子 ====================
def build_M(epsilon, seed=42):
    """构建11x11非厄米图拉普拉斯算子 M(epsilon) = L_G + kappa"""
    M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    
    # 非对角邻接微扰项
    np.random.seed(seed)
    V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
    for i in range(N_TOTAL):
        for j in range(N_TOTAL):
            if i != j:
                M[i, j] = epsilon * abs(V[i, j])
    
    # 对角正则抽吸项：时空扇区0，真空扇区1/R_root²
    kappa = np.zeros(N_TOTAL, dtype=complex)
    kappa[idx_space] = 0.0
    kappa[idx_vacuum] = 1.0 / R_ROOT**2
    
    # 行和自平衡（图拉普拉斯标准形式）
    for i in range(N_TOTAL):
        M[i, i] = -np.sum(M[:, i]) + kappa[i]
    
    return M

# ==================== 单元2：特征值全范围扫描 ====================
def scan_eigenvalues(epsilon_range):
    all_evals = []
    for eps in epsilon_range:
        M = build_M(eps)
        evals = linalg.eigvals(M)
        evals = sorted(evals, key=lambda x: x.real)
        all_evals.append(evals)
    return all_evals

def plot_eigenvalue_flow(all_evals, epsilon_list):

    ax1.scatter([all_evals[-1][i].real for i in range(N_TOTAL)],

    ax1.set_xlabel('Re(λ)', fontsize=12)
    ax1.set_ylabel('Im(λ)', fontsize=12)
    ax1.set_title('Eigenvalue Flow in Complex Plane', fontsize=14, fontweight='bold')

# ==================== 单元3：EP3临界点精细扫描 ====================
def compute_det_Meff(epsilon):
    """舒尔补计算有效空间矩阵，验证EP3三迹归零"""
    M = build_M(epsilon)
    M_SS = M[:N_SPACE, :N_SPACE]
    M_SV = M[:N_SPACE, N_SPACE:]
    M_VS = M[N_SPACE:, :N_SPACE]
    L_G_VV = M[N_SPACE:, N_SPACE:]
    kappa_reg = (1.0 / R_ROOT**2) * np.eye(N_VACUUM)
    
    try:
        inv_term = np.linalg.inv(L_G_VV + kappa_reg)
        M_eff = M_SS - M_SV @ inv_term @ M_VS
        return np.linalg.det(M_eff), np.trace(M_eff), np.trace(M_eff @ M_eff)
    except np.linalg.LinAlgError:
        return None, None, None

def plot_ep3_convergence(eps_fine, dets, traces, trace_sqs):
    fig, axes = plt.subplots(1, 3, figsize=(18, 5))
    titles = ['det(M_eff)', 'Tr(M_eff)', 'Tr(M_eff²)']
    data = [dets, traces, trace_sqs]
    colors = ['blue', 'red', 'green']
    
    for ax, title, vals, color in zip(axes, titles, data, colors):
        real_vals = [v.real if v is not None else np.nan for v in vals]
        ax.plot(eps_fine, real_vals, color=color, linewidth=2)
        ax.axhline(y=0, color='gray', linestyle='--', alpha=0.5)
        ax.axvline(x=EPSILON_C, color='black', linestyle=':', linewidth=2, 
                   label=f'ε_c={EPSILON_C}')
        ax.set_xlabel('ε', fontsize=12)
        ax.set_ylabel(title, fontsize=12)
        ax.set_title(f'{title} at EP3 Critical Point', fontsize=13, fontweight='bold')
        ax.grid(True, alpha=0.3)
        ax.legend()
    
    plt.suptitle('EP3 Critical Point: Three Traces Simultaneously Vanish', 
                 fontsize=16, y=1.02, fontweight='bold')
    plt.tight_layout()
    plt.show()

# ==================== 单元4：双正交波包塌缩动力学 ====================
def simulate_collapse(epsilon, t_max=15.0, n_steps=1000, gamma=1.5):
    """非厄米投影测量下的波包塌缩模拟，双正交度规物理归一化"""
    M = build_M(epsilon)
    
    # 双正交基：左右特征向量
    evals, eigvecs_right = linalg.eig(M)
    eigvecs_left = linalg.inv(eigvecs_right).T.conj()
    
    # 非正交重叠度规 S = R†R
    S = eigvecs_right.T.conj() @ eigvecs_right
    
    # 测量投影算子（前3个时空本征态为观测通道）
    P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for k in range(N_SPACE):
        P_meas += np.outer(eigvecs_right[:, k], eigvecs_left[:, k].conj())
    
    # 消相干有效哈密顿量：未观测子空间获得虚部阻尼
    Q_unobs = np.eye(N_TOTAL, dtype=complex) - P_meas
    M_collapse = M - 1j * gamma * Q_unobs
    
    # 时间演化
    t_array = np.linspace(0, t_max, n_steps)
    prob_unobs = np.zeros(n_steps)
    psi0 = np.ones(N_TOTAL, dtype=complex) / np.sqrt(N_TOTAL)  # 初始均匀叠加态
    
    for idx, t in enumerate(t_array):
        U_t = linalg.expm(-1j * M_collapse * t)
        psi_t = U_t @ psi0
        norm_physical = np.dot(psi_t.conj(), S @ psi_t).real
        prob_unobs[idx] = np.dot(psi_t.conj(), S @ Q_unobs @ psi_t).real / norm_physical
    
    return t_array, prob_unobs, S

def plot_collapse_dynamics(t_arr, prob_unobs):
    fig, axes = plt.subplots(1, 2, figsize=(16, 6))
    
    # 左图：线性坐标塌缩曲线
    ax1 = axes[0]
    ax1.plot(t_arr, prob_unobs, 'b-', linewidth=2.5, label='Unobserved state probability')
    ax1.axhline(y=np.exp(-1), color='red', linestyle='--', label='1/e threshold')
    
    tau_collapse = None
    tau_idx = np.where(prob_unobs <= np.exp(-1))[0]
    if len(tau_idx) > 0:
        tau_collapse = t_arr[tau_idx[0]]
        ax1.axvline(x=tau_collapse, color='green', linestyle=':', linewidth=2,
                   label=f'τ_collapse ≈ {tau_collapse:.2f} τ_P')
        ax1.scatter(tau_collapse, np.exp(-1), color='green', s=60, zorder=5)
        
    ax1.set_xlabel('Time t (Planck time τ_P)', fontsize=12)
    ax1.set_ylabel('Relative probability measure', fontsize=12)
    ax1.set_title('Wavefunction Collapse Dynamics (Linear Scale)', fontsize=13, fontweight='bold')
    ax1.grid(True, alpha=0.3)
    ax1.legend(fontsize=10)
    
    # 右图：对数坐标衰减速率
    ax2 = axes[1]
    ax2.plot(t_arr, prob_unobs, 'g-', linewidth=2.5, label='Decay trajectory')
    ax2.set_yscale('log')
    ax2.set_xlabel('Time t (τ_P)', fontsize=12)
    ax2.set_ylabel('Log probability', fontsize=12)
    ax2.set_title('Exponential Decay Rate (Log Scale)', fontsize=13, fontweight='bold')
    ax2.grid(True, which="both", alpha=0.3)
    ax2.legend(fontsize=10)
    
    plt.suptitle('Wavefunction Collapse under Non-Hermitian Projection', 
                 fontsize=16, y=1.02, fontweight='bold')
    plt.tight_layout()
    plt.show()
    return tau_collapse

# ==================== 单元6：δ_loop拓扑常数验证（修正版） ====================
def verify_delta_loop(epsilon=0.01):
    """弱场极限下验证拓扑边界纠缠比 δ_loop = 11/544"""
    M = build_M(epsilon)
    evals, eigvecs_right = linalg.eig(M)
    S = eigvecs_right.T.conj() @ eigvecs_right
    
    # 取3维观测子空间的度规偏离（对应边界纠缠修正量）
    S_space = S[:N_SPACE, :N_SPACE]
    delta_measured = np.mean(np.abs(S_space - np.eye(N_SPACE)))
    
    print("\n" + "="*60)
    print("【弱场拓扑边界纠缠比验证】")
    print("="*60)
    print(f"验证条件：弱场近似 ε = {epsilon}")
    print(f"数值子空间偏离度：{delta_measured:.6f}")
    print(f"理论值 δ_loop = 11/544 ≈ {DELTA_LOOP:.6f}")
    print(f"相对偏差：{abs(delta_measured - DELTA_LOOP)/DELTA_LOOP*100:.2f}%")
    print("="*60)
    return delta_measured
