采用体系标准的$11 \times 11$非厄米图拉普拉斯矩阵$\mathbf{M}$：

P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
for k in range(N_SPACE):
P_meas += np.outer(右矢[:, k], 左矢[:, k].conj())
I_mat = np.eye(N_TOTAL, dtype=complex)
Q_unobs = I_mat - P_meas
gamma_eff = gamma * DELTA_LOOP_THEORY
M_collapse = M - 1j * gamma_eff * Q_unobs
return M_collapse, Q_unobs

def 提取Kraus算子(epsilon, t, gamma=1.0):
"""
使用SVD确定平移因子，构造严格收缩的传播子，再提取Kraus算子
"""
M_collapse, _ = 获取塌缩算子(epsilon, gamma)
U_raw = linalg.expm(-1j * M_collapse * t)

# 计算U_raw的最大奇异值
s = linalg.svdvals(U_raw)
s_max = np.max(s)

# 确定平移因子：使稳定后最大奇异值为1
if s_max > 1.0:
    eta = np.log(s_max) / t
else:
    eta = 0.0

# 稳定后的传播子（严格收缩）
U = np.exp(-eta * t) * U_raw

# 缺陷矩阵 D = I - U†U
I_mat = np.eye(N_TOTAL, dtype=complex)
D = I_mat - U.conj().T @ U

# 特征分解D（此时应半正定）
特征值, 特征向量 = linalg.eigh(D)

# 滤除数值噪声（负特征值置零）
特征值 = np.maximum(特征值, 0.0)
阈值 = 1e-14
有效索引 = np.where(特征值 > 阈值)[0]

Kraus列表 = [U]  # A_0

# 标准Kraus构造：A_k = sqrt(λ_k) * |v_k><v_k|
for idx in 有效索引:
    val = 特征值[idx]
    vec = 特征向量[:, idx]
    A_k = np.sqrt(val) * np.outer(vec, vec.conj())
    Kraus列表.append(A_k)

return Kraus列表, eta

def 验证迹守恒(Kraus列表):
求和 = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
for A in Kraus列表:
求和 += A.conj().T @ A
I_mat = np.eye(N_TOTAL, dtype=complex)
误差 = linalg.norm(求和 - I_mat)
return 求和, 误差

def build_M(epsilon):
M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
np.random.seed(42)
V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
