def build_M(epsilon):
    """
    构建 11x11 非厄米图拉普拉斯算子
    M(epsilon) = L_G(epsilon) + kappa
    """
    M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    
    # 非对角微扰项
    np.random.seed(42)
    V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
    
    for i in range(N_TOTAL):
        for j in range(N_TOTAL):
            if i != j:
                M[i, j] = epsilon * abs(V[i, j])
    
    # 对角正则化抽吸算子
    kappa = np.zeros(N_TOTAL, dtype=complex)
    kappa[idx_space] = 0.0
    kappa[idx_vacuum] = 1.0 / R_ROOT**2
    
    # 行和自平衡
    for i in range(N_TOTAL):
        M[i, i] = -np.sum(M[:, i]) + kappa[i]
    
    return M

def scan_eigenvalues(epsilon_range):
    all_evals = []
    epsilon_list = []
    
    for eps in epsilon_range:
        M = build_M(eps)
        evals = linalg.eigvals(M)
        evals = sorted(evals, key=lambda x: x.real)
        all_evals.append(evals)
        epsilon_list.append(eps)
    
    return all_evals, epsilon_list

单元5：计算有效矩阵 $\mathbf{M}_{\text{eff}}$ 的行列式

def compute_det_Meff(epsilon):
    M = build_M(epsilon)
    
    M_SS = M[:N_SPACE, :N_SPACE]
    M_SV = M[:N_SPACE, N_SPACE:]
    M_VS = M[N_SPACE:, :N_SPACE]
    L_G_VV = M[N_SPACE:, N_SPACE:]
    
    kappa_reg = (1.0 / R_ROOT**2) * np.eye(N_VACUUM)
    
    try:
        inv_term = np.linalg.inv(L_G_VV + kappa_reg)
        M_eff = M_SS - M_SV @ inv_term @ M_VS
        det_val = np.linalg.det(M_eff)
        trace_val = np.trace(M_eff)
        trace_sq_val = np.trace(M_eff @ M_eff)
        return det_val, trace_val, trace_sq_val
    except np.linalg.LinAlgError:
        return None, None, None

eps_fine = np.linspace(EPSILON_C - 0.015, EPSILON_C + 0.01, 300)
dets, traces, trace_sqs = [], [], []

for eps in eps_fine:
    d, t, ts = compute_det_Meff(eps)

def simulate_collapse_perfect(epsilon, t_max=12.0, n_steps=600, gamma=2.0):
    """
    使用非厄米测量消相干算子 M - i*gamma*(I - P_meas) 精确模拟波包塌缩动力学
    
    修正1：测量哈密顿量改为 M_collapse = M - i*gamma*(I - P_meas)
    修正2：使用非正交度规 S 进行物理归一化
    
    参数:
        epsilon: 微扰参数
        t_max: 最大演化时间（普朗克时间单位）
        n_steps: 时间步数
        gamma: 测量强度（仪器与系统的耦合率）
    """
    M = build_M(epsilon)
    
    # 1. 计算右特征向量
    evals, eigvecs_right = linalg.eig(M)
    
    # 2. 计算左特征向量
    eigvecs_left = linalg.inv(eigvecs_right).T.conj()
    
    # 3. 构建非正交重叠度规矩阵 S_ij = <r_i|r_j>
    S = eigvecs_right.T.conj() @ eigvecs_right
    
    # 4. 定义测量投影算子 P_meas (选择前3个特征子空间作为观测通道)
    n_measured = N_SPACE
    P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for k in range(n_measured):
        P_meas += np.outer(eigvecs_right[:, k], eigvecs_left[:, k].conj())
    
    # 5. 构建消相干测量算子: M_collapse = M - i * gamma * (I - P_meas)
    I_mat = np.eye(N_TOTAL, dtype=complex)
    Q_unobs = I_mat - P_meas  # 未观测子空间的投影算子
    M_collapse = M - 1j * gamma * Q_unobs
    
    # 6. 时间动力学演化
    t_array = np.linspace(0, t_max, n_steps)
    prob_unobs = np.zeros(n_steps)
    
    # 初始状态：在所有本征态上均匀叠加
    psi0 = np.ones(N_TOTAL, dtype=complex) / np.sqrt(N_TOTAL)
    
    for idx, t in enumerate(t_array):
        # 求解非厄米薛定谔方程: d/dt |psi> = -i * M_collapse * |psi>
        U_t = linalg.expm(-1j * M_collapse * t)
        psi_t = U_t @ psi0
        
        # 使用非正交度规 S 计算物理归一化分母: <psi_t|S|psi_t>
        norm_physical = np.dot(psi_t.conj(), S @ psi_t).real
        
        # 计算未观测子空间在度规 S 下的物理期望值
        numerator = np.dot(psi_t.conj(), S @ Q_unobs @ psi_t).real
        prob_unobs[idx] = numerator / norm_physical
        
    return t_array, prob_unobs, S

def plot_collapse_dynamics_perfect(t_arr, prob_unobs, save_path=None):
    """
    绘制符合非厄米物理度规的波包塌缩实时演化曲线（对数坐标与线性坐标对比）
    """
    fig, axes = plt.subplots(1, 2, figsize=(16, 6))
    
    # ---- 左图：线性坐标下的塌缩过程 ----
    ax1 = axes[0]
    ax1.plot(t_arr, prob_unobs, 'b-', linewidth=2.5, label='未观测态物理概率 $\\mathcal{P}_{unobs}$')
    ax1.axhline(y=np.exp(-1), color='red', linestyle='--', label='$1/e$ 临界相消线')
    
    # 寻找 1/e 塌缩特征时间
    tau_idx = np.where(prob_unobs <= np.exp(-1))[0]
    if len(tau_idx) > 0:
        tau_collapse = t_arr[tau_idx[0]]
        ax1.axvline(x=tau_collapse, color='green', linestyle=':', linewidth=2,
                   label=f'$\\tau_{{collapse}} \\approx {tau_collapse:.2f} \\ \\tau_P$')
        ax1.scatter(tau_collapse, np.exp(-1), color='green', s=60, zorder=5)
        
    ax1.set_xlabel('时间 $t$ (普朗克时间单位 $\\tau_P$)', fontsize=12)
    ax1.set_ylabel('相对概率测度', fontsize=12)
