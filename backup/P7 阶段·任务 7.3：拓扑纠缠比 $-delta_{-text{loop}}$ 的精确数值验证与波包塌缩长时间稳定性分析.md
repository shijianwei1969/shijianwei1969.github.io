### 单元 2：构建控制算子 $\mathbf{M}(\epsilon)$

```python
def build_M(epsilon):
    """
    构建 11x11 非厄米图拉普拉斯算子 M。
    包含列和自平衡约束与真空汇正则项。
    """
    M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    np.random.seed(42)  # 固定种子保证可重复性
    V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
    
    # 非对角跃迁项
    for i in range(N_TOTAL):
        for j in range(N_TOTAL):
            if i != j:
                M[i, j] = epsilon * abs(V[i, j])
    
    # 对角汇项（时空扇区为 0，真空扇区为 1/R^2）
    kappa = np.zeros(N_TOTAL, dtype=complex)
    kappa[:N_SPACE] = 0.0
    kappa[N_SPACE:] = 1.0 / R_ROOT**2
    
    # 列和自平衡约束
    for i in range(N_TOTAL):
        M[i, i] = -np.sum(M[:, i]) + kappa[i]
    
    return M
```

### 单元 3：搜索亚临界稳定运行点 $\epsilon_{\text{loop}}$

```python
def compute_nonorthogonality_metric(epsilon):
    """
    计算非正交度规矩阵 S = R^† R，
    并按时空局域化权重排序特征向量，
    提取跨扇区最大非对角元。
    """
    M = build_M(epsilon)
    
    # 计算右特征向量
    evals, eigvecs_right = linalg.eig(M)
    
    # 按时空局域化权重排序（前 3 个为时空扇区）
    spatial_weights = np.sum(np.abs(eigvecs_right[:N_SPACE, :])**2, axis=0)
    sort_idx = np.argsort(spatial_weights)[::-1]
    eigvecs_right = eigvecs_right[:, sort_idx]
    
    # 列归一化
    for k in range(N_TOTAL):
        eigvecs_right[:, k] /= linalg.norm(eigvecs_right[:, k])
    
    # 计算左特征向量（双正交 companion）
    eigvecs_left = linalg.inv(eigvecs_right).T.conj()
    
    # 构建非正交重叠度规矩阵 S
    S = eigvecs_right.T.conj() @ eigvecs_right
    
    # 提取跨扇区块 S[:3, 3:] 的最大绝对值
    cross_block = np.abs(S[:N_SPACE, N_SPACE:])
    max_off_diag = np.max(cross_block)
    
    # 计算相对偏差
    relative_deviation = abs(max_off_diag - DELTA_LOOP_THEORY) / DELTA_LOOP_THEORY
    
    return S, max_off_diag, relative_deviation, eigvecs_right, eigvecs_left


def find_stable_operating_point():
    """
    在 [0, epsilon_c] 区间内二分搜索，
    使跨扇区最大非对角元等于 delta_loop。
    """
    low, high = 0.0, EPSILON_C
    for _ in range(100):
        mid = (low + high) / 2
        _, max_off, _, _, _ = compute_nonorthogonality_metric(mid)
        if max_off < DELTA_LOOP_THEORY:
            low = mid
        else:
            high = mid
    return low
```

### 单元 5：构建测量消相干算子并进行长时间演化

```python
def simulate_long_time_collapse(epsilon, t_max=600.0, n_steps=6000, gamma=1.0):
    """
    长时间波包塌缩动力学模拟（B 路线：保留 S 度规加权）。
    
    参数:
        epsilon: 微扰参数（应使用 epsilon_loop）
        t_max: 最大演化时间（普朗克时间单位 tau_P）
        n_steps: 时间步数
        gamma: 测量强度
    """
    M = build_M(epsilon)
    
    # 计算特征向量与度规
    S, _, _, eigvecs_right, eigvecs_left = compute_nonorthogonality_metric(epsilon)
    
    # 测量投影算子（前 3 个时空模式作为观测通道）
    n_measured = N_SPACE
    P_meas = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    for k in range(n_measured):
        P_meas += np.outer(eigvecs_right[:, k], eigvecs_left[:, k].conj())
    
    I_mat = np.eye(N_TOTAL, dtype=complex)
    Q_unobs = I_mat - P_meas  # 未观测子空间投影（真空背景）
    
    # 振幅时钟：gamma_eff = gamma * delta_loop
    gamma_eff = gamma * DELTA_LOOP_THEORY
    M_collapse = M - 1j * gamma_eff * Q_unobs
    
    # 时间网格
    t_array = np.linspace(0, t_max, n_steps)
    dt = t_array[1] - t_array[0]
    
    prob_unobs = np.zeros(n_steps)
    
    # 初始状态：均匀叠加
    psi_t = np.ones(N_TOTAL, dtype=complex) / np.sqrt(N_TOTAL)
    
    # 预计算单步传播子
    U_dt = linalg.expm(-1j * M_collapse * dt)
    
    # 快速步进循环
    for idx in range(n_steps):
        # S 加权物理概率
        norm_phys = np.dot(psi_t.conj(), S @ psi_t).real
        numerator = np.dot(psi_t.conj(), S @ Q_unobs @ psi_t).real
        prob_unobs[idx] = numerator / max(norm_phys, 1e-15)
        
        if idx < n_steps - 1:
            psi_t = U_dt @ psi_t
    
    return t_array, prob_unobs, S


# 执行长时间模拟（使用 epsilon_loop 和 gamma=1.0）
```

```python
def extract_collapse_time(t_arr, prob_unobs):
    """
    提取塌缩时间 tau_collapse（概率降到初始值 1/e 的时间）
    并计算指数衰减率 lambda。
    """
    # 相对 1/e 阈值
    threshold = np.exp(-1) * prob_unobs[0]
    idx_threshold = np.where(prob_unobs <= threshold)[0]
    
    if len(idx_threshold) > 0:
        idx_cross = idx_threshold[0]
        tau_collapse = t_arr[idx_cross]
        
        # 拟合窗口：取前 1.5 倍塌缩时间内的数据
        fit_end = max(15, min(len(t_arr), int(1.5 * idx_cross)))
        t_fit = t_arr[:fit_end]
        log_prob = np.log(np.maximum(prob_unobs[:fit_end], 1e-15))
        
        # 线性拟合：log(P) = -lambda * t + const
        coeffs = np.polyfit(t_fit, log_prob, 1)
        decay_rate = -coeffs[0]  # lambda
    else:
        tau_collapse = None
        decay_rate = None
    
    return tau_collapse, decay_rate
```
