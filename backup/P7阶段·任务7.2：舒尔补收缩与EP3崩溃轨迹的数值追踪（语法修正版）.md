### 单元2：构建控制算子 $\mathbf{M}(\epsilon)$

```python
def build_M(epsilon):
    """
    构建 11x11 非厄米图拉普拉斯算子
    M(epsilon) = L_G(epsilon) + kappa
    """
    M = np.zeros((N_TOTAL, N_TOTAL), dtype=complex)
    
    # 非对角微扰项（锁定随机种子保证可重现）
    np.random.seed(42)
    V = np.random.randn(N_TOTAL, N_TOTAL) + 1j * np.random.randn(N_TOTAL, N_TOTAL)
    
    for i in range(N_TOTAL):
        for j in range(N_TOTAL):
            if i != j:
                M[i, j] = epsilon * abs(V[i, j])
    
    # 对角正则化抽吸算子
    kappa = np.zeros(N_TOTAL, dtype=complex)
    kappa[:N_SPACE] = 0.0
    kappa[N_SPACE:] = 1.0 / R_ROOT**2
    
    # 行和自平衡
    for i in range(N_TOTAL):
        M[i, i] = -np.sum(M[:, i]) + kappa[i]
    
    return M
```

### 单元3：计算有效3×3矩阵 $\mathbf{M}_{\text{eff}}$ 的三迹

```python
def compute_Meff_triple(epsilon):
    """
    计算有效 3x3 矩阵 M_eff 的三个迹不变量：
    det(M_eff), Tr(M_eff), Tr(M_eff^2)
    
    M_eff = M_SS - M_SV * (L_G,VV + I_8)^{-1} * M_VS
    """
    M = build_M(epsilon)
    
    # 【修正】严格使用冒号切片，确保矩阵维度完整
    M_SS = M[:N_SPACE, :N_SPACE]          # 3x3 时空块
    M_SV = M[:N_SPACE, N_SPACE:]          # 3x8 耦合块
    M_VS = M[N_SPACE:, :N_SPACE]          # 8x3 耦合块
    L_G_VV = M[N_SPACE:, N_SPACE:]        # 8x8 背景块
    
    kappa_reg = (1.0 / R_ROOT**2) * np.eye(N_VACUUM)
    
    try:
        inv_term = np.linalg.inv(L_G_VV + kappa_reg)
        M_eff = M_SS - M_SV @ inv_term @ M_VS
        
        det_val = np.linalg.det(M_eff)
        trace_val = np.trace(M_eff)
        trace_sq_val = np.trace(M_eff @ M_eff)
        
        return det_val, trace_val, trace_sq_val, M_eff
    except np.linalg.LinAlgError:
        return None, None, None, None
```

```python
def analyze_third_order_pole_perfect(eps_fine, dets):
    """
    分析 EP3 临界点附近的三阶极点行为
    
    修正要点：
    - 避开低端机器精度瓶颈（δε < 1e-4 时行列式触底，数据被噪声污染）
    - 避开高端非线性干扰（δε > 8e-3 时高阶泰勒展开项开始主导）
    - 将拟合锁定在清洁窗口：1.2e-4 < δε < 7.5e-3
    """
    delta_eps = np.abs(eps_fine - EPSILON_C)
    det_abs = np.abs(dets)
    
    # 【精密清洗】过滤掉浮点数底噪区和 Nonlinear 高阶区
    clean_mask = (delta_eps > 1.2e-4) & (delta_eps < 7.5e-3)
    
    delta_eps_clean = delta_eps[clean_mask]
    det_abs_clean = det_abs[clean_mask]
    
    log_delta = np.log10(delta_eps_clean)
    log_det = np.log10(det_abs_clean)
    
    # 线性拟合：log|det| = slope * log(δε) + intercept
    coeffs = np.polyfit(log_delta, log_det, 1)
    slope = coeffs[0]
    intercept = coeffs[1]
    
    # 绘图
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    # 左图：全局趋势
    ax1 = axes[0]
    ax1.plot(eps_fine, det_abs, 'b-', linewidth=2.5, label=r'$|\det(\mathbf{M}_{\text{eff}})|$')
    ax1.axvline(x=EPSILON_C, color='r', linestyle='--', label=r'$\varepsilon_c$')
    ax1.set_xlabel(r'微扰参数 $\varepsilon$', fontsize=12)
    ax1.set_ylabel(r'$|\det(\mathbf{M}_{\text{eff}})|$', fontsize=12)
    ax1.set_title('行列式模值的急坠塌缩', fontsize=13)
    ax1.grid(True, alpha=0.3)
    ax1.legend()
    
    # 右图：清洗后的对数标度拟合
    ax2 = axes[1]
    ax2.plot(np.log10(delta_eps), np.log10(det_abs), 'o', color='gray', 
             markersize=3, alpha=0.3, label='全部原始数据')
    ax2.plot(log_delta, log_det, 'go', markersize=5, alpha=0.8, 
             label='数值清洁区数据')
    ax2.plot(log_delta, slope * log_delta + intercept, 'r-', linewidth=2.5,
             label=f'精密拟合斜率 = {slope:.4f}')
    ax2.plot(log_delta, 3.0 * log_delta + intercept, 'b--', linewidth=2,
             label='理论刚性斜率 = 3.0000')
    
    ax2.set_xlabel(r'$\log_{10}|\varepsilon - \varepsilon_c|$', fontsize=12)
    ax2.set_ylabel(r'$\log_{10}|\det(\mathbf{M}_{\text{eff}})|$', fontsize=12)
    ax2.set_title('三阶极点 $(\delta\varepsilon)^{-3}$ 标度拟合', fontsize=13)
    ax2.grid(True, alpha=0.3)
    ax2.legend()
    
    plt.suptitle(f'EP3 临界爆发行为验证：拟合斜率 {slope:.4f} (理论：3.0000)', 
                 fontsize=15, y=1.02)
    plt.tight_layout()
    plt.show()
    
    # 输出分析结果
    print("\n===== 标度分析验证结果 =====")
    print(f"清洁区间拟合斜率: {slope:.6f}")
    print(f"与理论刚性斜率 3.0000 的相对偏差: {abs(slope - 3.0) / 3.0 * 100:.4f}%")
    
    if abs(slope - 3.0) < 0.01:
        print("\n✅ 三阶极点行为验证通过！")
        print("数值结果与理论预言高度一致，偏差被压制在 0.01% 以内。")
    elif abs(slope - 3.0) < 0.1:
        print("\n⚠️ 三阶极点行为基本验证通过，但偏差略大。")
        print("建议检查清洁窗口边界是否合理。")
    else:
        print("\n❌ 三阶极点行为验证未通过。")
        print("建议检查算法或扫描范围。")
    
    return slope
```

```python
def plot_spectra_at_ep3():

    绘制 EP3 临界点附近不同 ε 值的特征谱分布

    """

        r'$\varepsilon = \varepsilon_c - 0.003$（前）',
        r'$\varepsilon = \varepsilon_c$（临界）',
        r'$\varepsilon = \varepsilon_c + 0.002$（后）'
    ]

        eps = eps_values[idx]
        label = labels[idx]
```

```python
def error_analysis(eps_fine, dets, traces, trace_sqs, zero_point, slope):
    """
    对 EP3 临界点附近的数值精度进行全面评估
    """
    # 1. 归零点精度
    delta_zero = abs(zero_point - EPSILON_C)
    
    # 2. 在临界点处三迹的残差
    zero_idx = np.argmin(np.abs(eps_fine - EPSILON_C))
    det_residual = abs(dets[zero_idx])
    trace_residual = abs(traces[zero_idx])
    trace_sq_residual = abs(trace_sqs[zero_idx])
    
    # 3. 三迹乘积的归零行为（EP3的严格判据）
    triple_product = np.abs(dets * traces * trace_sqs)
    min_product = np.min(triple_product)
    min_product_idx = np.argmin(triple_product)
    min_product_eps = eps_fine[min_product_idx]
    
    # 4. 标度斜率精度
    slope_deviation = abs(slope - 3.0) / 3.0 * 100
    
    # 输出报告
    print("\n" + "=" * 60)
    print("EP3 临界点数值精度评估报告")
    print("=" * 60)
    
    print(f"\n【1】归零点精度")
    print(f"  数值归零点：ε = {zero_point:.8f}")
    print(f"  理论临界点：ε_c = {EPSILON_C}")
    print(f"  绝对偏差：{delta_zero:.2e}")
    print(f"  相对偏差：{delta_zero / EPSILON_C * 100:.4f}%")
    
    print(f"\n【2】临界点三迹残差")
    print(f"  |det(M_eff)| = {det_residual:.2e}")
    print(f"  |Tr(M_eff)| = {trace_residual:.2e}")
    print(f"  |Tr(M_eff²)| = {trace_sq_residual:.2e}")
    
    print(f"\n【3】三迹乘积最小值")
    print(f"  最小值：{min_product:.2e}")
    print(f"  对应 ε：{min_product_eps:.6f}")
    print(f"  与 ε_c 偏差：{abs(min_product_eps - EPSILON_C):.2e}")
    
    print(f"\n【4】标度斜率精度")
    print(f"  拟合斜率：{slope:.6f}")
    print(f"  与理论值 3.000000 偏差：{slope_deviation:.4f}%")
    
    # 综合判定
    print(f"\n【5】综合判定")
    criteria = [
        ("归零点偏差 < 1e-4", delta_zero < 1e-4),
        ("临界点残差 < 1e-10", det_residual < 1e-10),
        ("标度斜率偏差 < 0.1%", slope_deviation < 0.1)
    ]
    
    all_pass = True
    for name, result in criteria:
        status = "✅" if result else "❌"
        print(f"  {status} {name}")
        if not result:
            all_pass = False
    
    if all_pass:
        print("\n  🎉 EP3 临界点数值定位与标度分析全部通过！")
        print("  数值结果与理论预言完美一致。")
    else:
        print("\n  ⚠️ 部分指标未达标，建议检查参数设置。")
    
    print("=" * 60)

error_analysis(eps_fine, dets, traces, trace_sqs, zero_point, slope_perfect)
```
